| CREATE OR REPLACE FUNCTION public.vincular_admin_como_profissional()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_profissional_id uuid;
  v_modelo_padrao_id uuid;
BEGIN
  -- Buscar o ID do profissional associado ao admin
  SELECT id INTO v_profissional_id
  FROM public.profissionais
  WHERE user_id = NEW.admin_id;

  -- Se o admin já tem cadastro de profissional, criar vínculo
  IF v_profissional_id IS NOT NULL THEN
    -- Buscar modelo de permissão padrão (primeiro modelo padrão disponível)
    SELECT id INTO v_modelo_padrao_id
    FROM public.modelos_permissao
    WHERE is_padrao = true
    ORDER BY created_at
    LIMIT 1;

    -- Criar vínculo entre profissional e estabelecimento
    INSERT INTO public.vinculo_profissional_estabelecimento (
      profissional_id,
      estabelecimento_id,
      modelo_permissao_id,
      ativo
    ) VALUES (
      v_profissional_id,
      NEW.id,
      v_modelo_padrao_id,
      true
    );
  END IF;

  RETURN NEW;
END;
$function$
;

| CREATE OR REPLACE FUNCTION public.tem_permissao(_user_id uuid, _estabelecimento_id uuid, _modulo modulo_sistema, _nivel_minimo nivel_permissao DEFAULT 'visualizar'::nivel_permissao)
 RETURNS boolean
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  -- Admin do estabelecimento tem todas as permissões
  SELECT CASE 
    WHEN EXISTS (
      SELECT 1 FROM estabelecimentos 
      WHERE id = _estabelecimento_id AND admin_id = _user_id
    ) THEN true
    -- Verificar permissões customizadas do profissional
    WHEN EXISTS (
      SELECT 1 
      FROM vinculo_profissional_estabelecimento v
      JOIN profissionais p ON v.profissional_id = p.id
      JOIN permissoes_profissional pp ON pp.vinculo_id = v.id
      WHERE p.user_id = _user_id 
        AND v.estabelecimento_id = _estabelecimento_id
        AND v.ativo = true
        AND pp.modulo = _modulo
        AND (
          (_nivel_minimo = 'visualizar') OR
          (_nivel_minimo = 'editar' AND pp.nivel IN ('editar', 'administrar')) OR
          (_nivel_minimo = 'administrar' AND pp.nivel = 'administrar')
        )
    ) THEN true
    -- Verificar permissões do modelo aplicado
    WHEN EXISTS (
      SELECT 1
      FROM vinculo_profissional_estabelecimento v
      JOIN profissionais p ON v.profissional_id = p.id
      JOIN modelos_permissao mp ON v.modelo_permissao_id = mp.id
      JOIN permissoes_modelo pm ON pm.modelo_id = mp.id
      WHERE p.user_id = _user_id
        AND v.estabelecimento_id = _estabelecimento_id
        AND v.ativo = true
        AND pm.modulo = _modulo
        AND (
          (_nivel_minimo = 'visualizar') OR
          (_nivel_minimo = 'editar' AND pm.nivel IN ('editar', 'administrar')) OR
          (_nivel_minimo = 'administrar' AND pm.nivel = 'administrar')
        )
    ) THEN true
    ELSE false
  END;
$function$
;
